<script>
  let globalDashboardData = null;
  let currentModalZone = 'ALL';

  // --- กำหนดค่า TARGET ---
  const OVERALL_TARGET_PERCENT = 0.80;                                    // Target รวมภาพรวม
  const MAIN_ZONES_COUNT = 3;                                             // จำนวน Zone หลัก (A, B, C)
  const ZONE_TARGET_PERCENT = OVERALL_TARGET_PERCENT / MAIN_ZONES_COUNT; // Target ราย Zone ≈ 0.27%

  window.onload = function() {
    loadDashboardData(null);
  };

  function showLoading(msg) {
    document.getElementById('loadingText').innerText = msg || 'กำลังโหลดข้อมูลระบบ...';
    document.getElementById('loadingOverlay').style.display = 'flex';
  }

  function hideLoading() {
    document.getElementById('loadingOverlay').style.display = 'none';
  }

  function loadDashboardData(selectedColIndex) {
    showLoading('กำลังดึงข้อมูลจาก Google Sheets...');
    if (typeof google !== 'undefined' && google.script && google.script.run) {
      google.script.run
        .withSuccessHandler(function(data) {
          renderDashboard(data);
          hideLoading();
        })
        .withFailureHandler(function(err) {
          showError('System Error: ' + err.message);
          hideLoading();
        })
        .getDashboardData(selectedColIndex);
    } else {
      hideLoading();
    }
  }

  function onDateChange() {
    let selectedCol = document.getElementById('dateSelect').value;
    if (selectedCol) {
      loadDashboardData(selectedCol);
    }
  }

  function renderDashboard(data) {
    if (data.error) {
      showError('Data Fetch Error: ' + data.error);
      return;
    }

    hideError();
    globalDashboardData = data;

    let dateSelect = document.getElementById('dateSelect');
    if (data.availableDates && data.availableDates.length > 0) {
      dateSelect.innerHTML = '';
      data.availableDates.forEach(function(item) {
        let opt = document.createElement('option');
        opt.value = item.colIndex;
        opt.text = item.dateText;
        if (item.colIndex == data.selectedColIndex) {
          opt.selected = true;
        }
        dateSelect.appendChild(opt);
      });
    } else {
      dateSelect.innerHTML = '<option value="">ไม่พบข้อมูลวันที่</option>';
    }

    document.getElementById('totalSkip').innerText = (data.totalSkipCases || 0).toLocaleString() + ' เคส';
    document.getElementById('machineCount').innerText = (data.machineCount || 0).toLocaleString();
    document.getElementById('systemCount').innerText = (data.systemCount || 0).toLocaleString();
    
    document.getElementById('actualVol').innerText = (data.socActual || 0).toLocaleString();
    
    let skipPercentVal = parseFloat(data.skipPercentage || 0);
    let skipPercentElem = document.getElementById('skipPercent');
    let targetStatusElem = document.getElementById('targetStatusText');
    let targetCardBorder = document.getElementById('targetCardBorder');
    let targetBadge = document.getElementById('targetBadge');

    skipPercentElem.innerText = (data.skipPercentage || '0.00') + ' %';

    if (skipPercentVal <= OVERALL_TARGET_PERCENT) {
      skipPercentElem.className = 'metric-value text-success';
      targetCardBorder.className = 'card card-metric p-3 bg-white border-start border-4 border-success';
      targetBadge.className = 'badge bg-success fw-bold';
      targetBadge.innerText = '🎯 Overall Target: < 0.80%';
      targetStatusElem.innerHTML = `<span class="badge bg-success">🟢 Passed</span> <span class="text-muted small">อยู่ในเกณฑ์ภาพรวม (<= 0.80%)</span>`;
    } else {
      skipPercentElem.className = 'metric-value text-danger';
      targetCardBorder.className = 'card card-metric p-3 bg-white border-start border-4 border-danger';
      targetBadge.className = 'badge bg-danger fw-bold';
      targetBadge.innerText = '🎯 Overall Target: < 0.80%';
      targetStatusElem.innerHTML = `<span class="badge bg-danger">🔴 Exceeded</span> <span class="text-muted small">เกินเป้าหมายภาพรวม (> 0.80%)</span>`;
    }

    renderSummaryTable(data);
  }

  function renderSummaryTable(data) {
    let tbody = document.getElementById('summaryTableBody');
    let skipCountByZone = data.skipCountByZone || {};
    let staffByZone = data.staffByZone || {};
    let socActual = data.socActual || 0;

    let targetZones = ['ALL', 'A', 'B', 'C', 'INTERSOC', 'RETURN'];
    
    let existingZones = Array.from(new Set([...Object.keys(skipCountByZone), ...Object.keys(staffByZone)]))
      .filter(zone => zone !== 'MS' && zone !== 'TBS');

    let allZones = Array.from(new Set([...targetZones, ...existingZones]));

    let sumABC = (skipCountByZone['A'] || 0) + (skipCountByZone['B'] || 0) + (skipCountByZone['C'] || 0);

    let allPercentVal = socActual > 0 ? (sumABC / socActual) * 100 : 0;

    let rowsHtml = '';
    allZones.forEach(function(zone) {
      let count = 0;
      let percentVal = 0;

      if (zone === 'ALL') {
        count = sumABC;
        percentVal = allPercentVal;
      } else {
        count = skipCountByZone[zone] || 0;
        percentVal = socActual > 0 ? (count / socActual) * 100 : 0;
      }

      let percentStr = percentVal.toFixed(2);
      let staffList = staffByZone[zone] || [];

      let staffBadges = staffList.length > 0 
        ? staffList.map(s => `<span class="staff-badge">👤 ${s}</span>`).join('')
        : '<span class="text-muted small">ไม่มีเจ้าหน้าที่</span>';

      let targetModalZone = (zone === 'ALL') ? 'ABC' : zone;

      let countDisplay = count > 0 
        ? `<span class="clickable-count" onclick="openRawModal('${targetModalZone}')" title="คลิกเพื่อดู Raw Data ${zone}">${count.toLocaleString()}</span>`
        : `<span class="text-muted">0</span>`;

      let rowBg = (zone === 'ALL') ? 'class="table-secondary fw-bold"' : '';

      let isZoneRed = false;

      if (zone === 'ALL') {
        isZoneRed = allPercentVal > OVERALL_TARGET_PERCENT;
      } else if (['A', 'B', 'C'].includes(zone)) {
        if (percentVal > ZONE_TARGET_PERCENT) {
          isZoneRed = true;
        }
      } else {
        if (percentVal > OVERALL_TARGET_PERCENT) {
          isZoneRed = true;
        }
      }

      let percentColorClass = isZoneRed ? 'text-danger' : 'text-success';

      rowsHtml += `<tr ${rowBg}>
        <td class="fw-bold text-dark">${zone}</td>
        <td class="text-center">${countDisplay}</td>
        <td class="text-center fw-bold ${percentColorClass}">${percentStr}%</td>
        <td>${staffBadges}</td>
      </tr>`;
    });

    tbody.innerHTML = rowsHtml;
  }

  // --- ระบบอัปโหลดแบบ BATCH / CHUNKING ---
  function handleCSVUpload(input) {
    let file = input.files[0];
    if (!file) return;

    showLoading('กำลังอ่านไฟล์ CSV และจัดเตรียมข้อมูล...');

    const TARGET_HEADERS = [
      'shipment_id', 'report_date', 'first_soc_received_operator_name',
      'first_soc_packed_operator_name', 'first_soc_lh_packed_operator_name',
      'soc_outbound_route_type', 'latest_to_number', 'is_in_sorting_center',
      'soc_outbound_late_type_2nd_cutoff', 'dest_station_name',
      'first_soc_intra_handover_packing_operator_name',
      'first_soc_intra_handover_packed_operator_name',
      'first_outbound_soc_staging_timestamp', 'recieve_team',
      'soc_outbound_based_received_2nd_cut_off_timestamp'
    ];

    Papa.parse(file, {
      header: true,
      skipEmptyLines: true,
      complete: function(results) {
        let rows = results.data;
        if (!rows || rows.length === 0) {
          hideLoading();
          alert('ไม่พบข้อมูลในไฟล์ CSV');
          return;
        }

        let formattedData = rows.map(row => {
          return TARGET_HEADERS.map(header => row[header] !== undefined ? row[header] : '');
        });

        showLoading('กำลังเคลียร์ข้อมูลเก่าใน Sheet...');
        google.script.run
          .withSuccessHandler(function(res) {
            if (res.success) {
              const CHUNK_SIZE = 2000;
              uploadCsvChunks(formattedData, 0, CHUNK_SIZE);
            } else {
              hideLoading();
              showError('เกิดข้อผิดพลาดในการเคลียร์ข้อมูล: ' + res.error);
            }
          })
          .withFailureHandler(function(err) {
            hideLoading();
            showError('Clear Sheet Fail: ' + err.message);
          })
          .clearRawDataSheet();
      }
    });
  }

  function uploadCsvChunks(allData, startIndex, chunkSize) {
    let totalRows = allData.length;
    let chunk = allData.slice(startIndex, startIndex + chunkSize);
    let isLastChunk = (startIndex + chunkSize) >= totalRows;
    let currentEnd = Math.min(startIndex + chunkSize, totalRows);

    showLoading(`กำลังบันทึกข้อมูล (${currentEnd.toLocaleString()} / ${totalRows.toLocaleString()} รายการ)...`);

    google.script.run
      .withSuccessHandler(function(res) {
        if (res.success) {
          if (!isLastChunk) {
            uploadCsvChunks(allData, startIndex + chunkSize, chunkSize);
          } else {
            showLoading('อัปโหลดสำเร็จ! กำลังโหลด Dashboard ใหม่...');
            document.getElementById('csvFileInput').value = '';
            loadDashboardData(null);
          }
        } else {
          hideLoading();
          showError('เกิดข้อผิดพลาดในการบันทึก: ' + res.error);
        }
      })
      .withFailureHandler(function(err) {
        hideLoading();
        showError('Upload Fail: ' + err.message);
      })
      .appendCsvChunkToRawSheet(chunk);
  }

  function openRawModal(targetZone) {
    if (!globalDashboardData || !globalDashboardData.skipCases) return;

    currentModalZone = targetZone || 'ALL';
    let skipCases = globalDashboardData.skipCases;
    let modalTitle = document.getElementById('rawDataModalLabel');

    let filteredCases = [];

    if (targetZone === 'ABC') {
      filteredCases = skipCases.filter(item => ['A', 'B', 'C'].includes(item.zone));
      modalTitle.innerText = `📄 รายการ Raw Data - Zone A, B, C (ทั้งหมด ${filteredCases.length.toLocaleString()} เคส)`;
    } else if (targetZone && targetZone !== 'ALL') {
      filteredCases = skipCases.filter(item => item.zone === targetZone);
      modalTitle.innerText = `📄 รายการ Raw Data - Zone ${targetZone} (ทั้งหมด ${filteredCases.length.toLocaleString()} เคส)`;
    } else {
      filteredCases = skipCases;
      modalTitle.innerText = `📄 รายการ Raw Data ทั้งหมด (ทั้งหมด ${skipCases.length.toLocaleString()} เคส)`;
    }

    renderRawModalBody(filteredCases);

    document.getElementById('searchRaw').value = '';
    let modal = new bootstrap.Modal(document.getElementById('rawDataModal'));
    modal.show();
  }

  function renderRawModalBody(skipCases) {
    let tbody = document.getElementById('rawTableBody');
    if (!skipCases || skipCases.length === 0) {
      tbody.innerHTML = '<tr><td colspan="5" class="text-center text-muted p-3">ไม่พบรายการ Skip Process ใน Zone นี้</td></tr>';
      return;
    }

    let html = '';
    skipCases.forEach(function(item, idx) {
      let badgeClass = 'bg-danger';
      if (item.reason === 'skip_process_intersoc_machine') badgeClass = 'bg-warning text-dark';
      if (item.reason === 'skip_process_intersoc_system') badgeClass = 'bg-info text-dark';

      html += `<tr>
        <td>${idx + 1}</td>
        <td class="fw-bold text-primary">${item.shipmentId}</td>
        <td><span class="badge ${badgeClass}">${item.reason}</span></td>
        <td><span class="badge bg-secondary">${item.zone}</span></td>
        <td class="fw-bold text-dark">${item.hub}</td>
      </tr>`;
    });
    tbody.innerHTML = html;
  }

  function filterRawTable() {
    let input = document.getElementById("searchRaw").value.toUpperCase();
    let trs = document.getElementById("rawTableBody").getElementsByTagName("tr");

    for (let i = 0; i < trs.length; i++) {
      let text = trs[i].innerText.toUpperCase();
      trs[i].style.display = (input === '' || text.indexOf(input) > -1) ? "" : "none";
    }
  }

  function exportCurrentModalToCSV() {
    let trs = document.getElementById("rawTableBody").getElementsByTagName("tr");
    if (!trs || trs.length === 0 || trs[0].getElementsByTagName("td").length <= 1) {
      alert("ไม่มีข้อมูลสำหรับ Export");
      return;
    }

    let csvData = [["No", "Shipment ID", "Skip Reason", "Zone", "Hub (Dest Station)"]];

    let rowNum = 1;
    for (let i = 0; i < trs.length; i++) {
      if (trs[i].style.display !== "none") {
        let tds = trs[i].getElementsByTagName("td");
        if (tds.length >= 5) {
          let shipmentId = tds[1].innerText.trim();
          let reason = tds[2].innerText.trim();
          let zone = tds[3].innerText.trim();
          let hub = tds[4].innerText.trim();

          csvData.push([rowNum++, shipmentId, reason, zone, hub]);
        }
      }
    }

    if (csvData.length <= 1) {
      alert("ไม่พบข้อมูลจากการค้นหาเพื่อ Export");
      return;
    }

    let csvContent = "\uFEFF";
    csvData.forEach(rowArray => {
      let row = rowArray.map(val => `"${String(val).replace(/"/g, '""')}"`).join(",");
      csvContent += row + "\r\n";
    });

    let blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
    let link = document.createElement("a");
    let url = URL.createObjectURL(blob);
    let fileName = `RawData_Zone_${currentModalZone}_${new Date().toISOString().slice(0, 10)}.csv`;

    link.setAttribute("href", url);
    link.setAttribute("download", fileName);
    link.style.visibility = 'hidden';
    document.body.appendChild(link);
    link.click();
    document.body.removeChild(link);
  }

  function showError(msg) {
    let errDiv = document.getElementById('errorAlert');
    errDiv.innerText = msg;
    errDiv.classList.remove('d-none');
  }

  function hideError() {
    let errDiv = document.getElementById('errorAlert');
    errDiv.classList.add('d-none');
  }

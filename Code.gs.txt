function corsJson(data) {
  return ContentService
    .createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}

// ============================================================
// doGet
// ============================================================
function doGet(e) {
  if (e && e.parameter && e.parameter.action) {
    const action = e.parameter.action;
    if (action === 'getFileList')   return corsJson(driveGetFileList());
    if (action === 'getLatestData') return corsJson(driveGetLatestData());
    if (action === 'getFileData')   return corsJson(driveGetFileData(e.parameter.filename));
    if (action === 'getDashboardData') return corsJson(getDashboardData(e.parameter.colIndex || null));
    return corsJson({ success: false, error: 'Unknown action: ' + action });
  }
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('SOC Investigation & Skip Process Dashboard')
    .setXFrameOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL);
}

function doPost(e) {
  try {
    const payload = JSON.parse(e.postData.contents);
    if (payload.action === 'saveFile') {
      return corsJson(driveSaveFile(payload.filename, payload.data));
    }
    return corsJson({ success: false, error: 'Unknown action' });
  } catch (err) {
    return corsJson({ success: false, error: err.toString() });
  }
}

function driveSaveFile(filename, data) {
  try {
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const safeBase = filename.replace(/[^a-zA-Z0-9._-]/g, '_');
    const jsonFilename = 'socn_' + safeBase + '.json';

    _deleteFileIfExists(folder, jsonFilename);

    const toSave = Object.assign({}, data);
    delete toSave.rawRows;
    folder.createFile(jsonFilename, JSON.stringify(toSave), MimeType.PLAIN_TEXT);

    _deleteFileIfExists(folder, 'socn_latest.json');
    const latestData = Object.assign({}, toSave, {
      _sourceFile: filename,
      _savedAt: new Date().toISOString()
    });
    folder.createFile('socn_latest.json', JSON.stringify(latestData), MimeType.PLAIN_TEXT);

    _updateFileIndex(folder, jsonFilename);

    return { success: true, savedAs: jsonFilename };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function driveGetFileList() {
  try {
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const indexFiles = folder.getFilesByName('socn_file_index.json');
    if (indexFiles.hasNext()) {
      return { success: true, files: JSON.parse(indexFiles.next().getBlob().getDataAsString()) };
    }
    return { success: true, files: [] };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function driveGetFileData(filename) {
  try {
    if (!filename) return { success: false, error: '??????????????????' };
    const folder = DriveApp.getFolderById(DRIVE_FOLDER_ID);
    const files = folder.getFilesByName(filename);
    if (files.hasNext()) {
      return { success: true, data: JSON.parse(files.next().getBlob().getDataAsString()) };
    }
    return { success: false, error: '?????????: ' + filename };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function _deleteFileIfExists(folder, filename) {
  const files = folder.getFilesByName(filename);
  while (files.hasNext()) files.next().setTrashed(true);
}

function _updateFileIndex(folder, newFilename) {
  try {
    let index = [];
    const indexFiles = folder.getFilesByName('socn_file_index.json');
    if (indexFiles.hasNext()) {
      const f = indexFiles.next();
      try { index = JSON.parse(f.getBlob().getDataAsString()); } catch(e) {}
      f.setTrashed(true);
    }
    index = index.filter(f => f !== newFilename);
    index.unshift(newFilename);
    if (index.length > 20) index = index.slice(0, 20);
    folder.createFile('socn_file_index.json', JSON.stringify(index), MimeType.PLAIN_TEXT);
  } catch(e) { Logger.log('_updateFileIndex error: ' + e); }
}

function getDashboardData(selectedDateColIndex) {
  try {
    let staffByZone = {};
    try {
      const rosterSs = SpreadsheetApp.openById(ROSTER_SPREADSHEET_ID);
      const rosterSheet = rosterSs.getSheetByName('FTE /Aug') || rosterSs.getSheets()[0];
      const rosterLastRow = rosterSheet.getLastRow();
      if (rosterLastRow >= 4) {
        const rosterValues = rosterSheet.getRange(4, 6, rosterLastRow - 3, 3).getValues();
        for (let i = 0; i < rosterValues.length; i++) {
          let name = rosterValues[i][0];
          let zone = rosterValues[i][2];
          if (name && zone) {
            zone = String(zone).trim().toUpperCase();
            if (!staffByZone[zone]) staffByZone[zone] = [];
            staffByZone[zone].push(String(name).trim());
          }
        }
      }
    } catch (e) { Logger.log("Roster read error: " + e); }

    let skipCases = [], skipCountByZone = {}, totalSkipCases = 0, machineCount = 0, systemCount = 0;
    try {
      const rawSs = SpreadsheetApp.openById(RAW_DATA_SPREADSHEET_ID);
      const rawSheet = rawSs.getSheetByName('Raw') || rawSs.getSheets()[0];
      const colAValues = rawSheet.getRange("A:A").getValues();
      let actualLastRow = 0;
      for (let i = colAValues.length - 1; i >= 0; i--) {
        if (colAValues[i][0] !== "" && colAValues[i][0] !== null) { actualLastRow = i + 1; break; }
      }
      const rawLastCol = rawSheet.getLastColumn();
      if (actualLastRow >= 2) {
        const headers = rawSheet.getRange(1, 1, 1, rawLastCol).getValues()[0].map(h => h ? String(h).trim().toLowerCase() : '');
        const shipmentIdx = headers.indexOf('shipment_id');
        const reasonIdx   = headers.indexOf('soc_outbound_late_type_2nd_cutoff');
        const hubIdx      = headers.indexOf('dest_station_name');
        const rawValues   = rawSheet.getRange(2, 1, actualLastRow - 1, rawLastCol).getValues();
        for (let i = 0; i < rawValues.length; i++) {
          let row = rawValues[i];
          let shipmentId = (shipmentIdx !== -1 && row[shipmentIdx]) ? String(row[shipmentIdx]).trim() : String(row[0] || '-').trim();
          let rawReason  = reasonIdx !== -1 ? String(row[reasonIdx] || '').trim().toLowerCase() : String(row[8] || row[18] || '').trim().toLowerCase();
          if (rawReason === 'undefined') rawReason = '';
          let hubName = (hubIdx !== -1 && row[hubIdx]) ? String(row[hubIdx]).trim() : String(row[9] || '-').trim();
          let zone = String(row[19] || row[9] || 'N/A').trim().toUpperCase();
          if (zone === 'UNDEFINED' || !zone) zone = 'N/A';
          if (rawReason.indexOf('skip') !== -1) {
            totalSkipCases++;
            if (rawReason.indexOf('machine') !== -1) machineCount++;
            else if (rawReason.indexOf('system') !== -1) systemCount++;
            skipCases.push({ rowNo: i+2, shipmentId: shipmentId||'-', reason: rawReason||'skip_outbound', zone, hub: hubName });
            if (zone) skipCountByZone[zone] = (skipCountByZone[zone] || 0) + 1;
          }
        }
      }
    } catch (e) { Logger.log("Raw data read error: " + e); }

    let availableDates = [], socActual = 0, targetCol = -1;
    try {
      const volSs = SpreadsheetApp.openById(VOLUME_SPREADSHEET_ID);
      const volSheet = volSs.getSheetByName('SOC') || volSs.getSheets()[0];
      const volLastCol = volSheet.getLastColumn();
      if (volLastCol >= 16) {
        const row1 = volSheet.getRange(1, 1, 1, volLastCol).getDisplayValues()[0];
        const row10 = volSheet.getRange(10, 1, 1, volLastCol).getValues()[0];
        for (let c = 15; c < row1.length; c++) {
          let dateStr = String(row1[c]).trim();
          if (dateStr !== "" && !dateStr.startsWith('#')) availableDates.push({ colIndex: c, dateText: dateStr });
        }
        targetCol = (selectedDateColIndex !== undefined && selectedDateColIndex !== null && selectedDateColIndex !== "")
          ? Number(selectedDateColIndex)
          : (availableDates.length > 0 ? availableDates[0].colIndex : -1);
        if (targetCol >= 0 && row10[targetCol] !== undefined) {
          let val = Number(row10[targetCol]);
          if (!isNaN(val)) socActual = val;
        }
      }
    } catch (e) { Logger.log("Volume read error: " + e); }

    return {
      staffByZone, skipCases, skipCountByZone,
      totalSkipCases, machineCount, systemCount,
      socActual, skipPercentage: socActual > 0 ? ((totalSkipCases / socActual) * 100).toFixed(2) : "0.00",
      availableDates, selectedColIndex: targetCol
    };
  } catch (err) {
    return { error: err.toString() };
  }
}

function clearRawDataSheet() {
  try {
    const rawSs = SpreadsheetApp.openById(RAW_DATA_SPREADSHEET_ID);
    const rawSheet = rawSs.getSheetByName('Raw') || rawSs.getSheets()[0];
    const maxRows = rawSheet.getMaxRows();
    if (maxRows >= 2) rawSheet.getRange(2, 1, maxRows - 1, 15).clearContent();
    SpreadsheetApp.flush();
    return { success: true };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

function appendCsvChunkToRawSheet(chunkData) {
  try {
    if (!chunkData || chunkData.length === 0) return { success: true, count: 0 };
    const rawSs = SpreadsheetApp.openById(RAW_DATA_SPREADSHEET_ID);
    const rawSheet = rawSs.getSheetByName('Raw') || rawSs.getSheets()[0];
    const colAValues = rawSheet.getRange("A:A").getValues();
    let actualLastRow = 0;
    for (let i = colAValues.length - 1; i >= 0; i--) {
      if (colAValues[i][0] !== "" && colAValues[i][0] !== null) { actualLastRow = i + 1; break; }
    }
    const targetStartRow = Math.max(actualLastRow + 1, 2);
    rawSheet.getRange(targetStartRow, 1, chunkData.length, chunkData[0].length).setValues(chunkData);
    SpreadsheetApp.flush();
    return { success: true, count: chunkData.length };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// ========== API Key สำหรับ doPost (เว็บ Server ใช้ key นี้ยิงมา) ==========
const API_SECRET_KEY = 'SOCN_OBBL_2026_SECRET_KEY_XK9M3';

// ========== doGet: ใช้งาน Web App UI เดิม ==========
function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('SOC Investigation & Skip Process Dashboard')

}

// ========== doPost: รับ Request จากเว็บ Server (ใช้ API Key ลับ) ==========
function doPost(e) {
  try {
    const params = e && e.postData && e.postData.contents
      ? JSON.parse(e.postData.contents)
      : {};

    // ตรวจ API Key
    if (!params.key || params.key !== API_SECRET_KEY) {
      return buildJsonResponse({ success: false, error: 'Unauthorized: Invalid API Key' });
    }

    const page = (params.page || '').toLowerCase();

    if (page === 'obbl') {
      return buildJsonResponse(getObBlDataForApi());
    } else {
      return buildJsonResponse({ success: false, error: 'Unknown page: ' + page });
    }

  } catch(err) {
    return buildJsonResponse({ success: false, error: err.toString() });
  }
}

function buildJsonResponse(obj) {
  const output = ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
  return output;
}

// ========== ดึงข้อมูล OB BL จาก RAW_DATA_SPREADSHEET ==========
function getObBlDataForApi() {
  try {
    const rawSs = SpreadsheetApp.openById(RAW_DATA_SPREADSHEET_ID);
    
    // หา sheet ที่ชื่อ OB หรือ OBBL หรือ sheet แรก
    let rawSheet = rawSs.getSheetByName('OB BL')
      || rawSs.getSheetByName('OBBL')
      || rawSs.getSheetByName('OB_BL')
      || rawSs.getSheetByName('Raw')
      || rawSs.getSheets()[0];

    const lastRow = rawSheet.getLastRow();
    const lastCol = rawSheet.getLastColumn();

    if (lastRow < 2 || lastCol < 1) {
      return { success: false, error: 'ไม่พบข้อมูลใน Sheet' };
    }

    // ดึง Headers
    const headers = rawSheet.getRange(1, 1, 1, lastCol).getValues()[0]
      .map(h => h ? String(h).trim() : '');

    // หาบรรทัดสุดท้ายที่มีข้อมูลจริงใน Col A
    const colAValues = rawSheet.getRange('A:A').getValues();
    let actualLastRow = 1;
    for (let i = colAValues.length - 1; i >= 1; i--) {
      if (colAValues[i][0] !== '' && colAValues[i][0] !== null && colAValues[i][0] !== undefined) {
        actualLastRow = i + 1;
        break;
      }
    }

    const dataCount = actualLastRow - 1;
    if (dataCount <= 0) {
      return { success: false, error: 'ไม่มีข้อมูลในแถวที่ 2 เป็นต้นไป' };
    }

    const rawValues = rawSheet.getRange(2, 1, dataCount, lastCol).getValues();

    // Convert rows เป็น Array of Arrays
    const rows = rawValues.map(row =>
      row.map(cell => {
        if (cell instanceof Date) {
          return Utilities.formatDate(cell, Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss');
        }
        return cell === null || cell === undefined ? '' : cell;
      })
    );

    return {
      success: true,
      headers: headers,
      rows: rows,
      total: rows.length,
      sheet: rawSheet.getName(),
      generatedAt: Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'yyyy-MM-dd HH:mm:ss')
    };

  } catch(err) {
    return { success: false, error: 'getObBlDataForApi error: ' + err.toString() };
  }
}

function getDashboardData(selectedDateColIndex) {
  try {
    // 1. Roster Sheet
    let staffByZone = {};
    try {
      const rosterSs = SpreadsheetApp.openById(ROSTER_SPREADSHEET_ID);
      const rosterSheet = rosterSs.getSheetByName('FTE /Aug') || rosterSs.getSheets()[0];
      const rosterLastRow = rosterSheet.getLastRow();
      
      if (rosterLastRow >= 4) {
        const rosterValues = rosterSheet.getRange(4, 6, rosterLastRow - 3, 3).getValues();
        for (let i = 0; i < rosterValues.length; i++) {
          let name = rosterValues[i][0]; 
          let zone = rosterValues[i][2]; 
          if (name && zone) {
            zone = String(zone).trim().toUpperCase();
            if (!staffByZone[zone]) staffByZone[zone] = [];
            staffByZone[zone].push(String(name).trim());
          }
        }
      }
    } catch (e) {
      Logger.log("Roster read error: " + e.toString());
    }

    // 2. Raw Data Sheet
    let skipCases = [];
    let skipCountByZone = {};
    let totalSkipCases = 0;
    let machineCount = 0;
    let systemCount = 0;

    try {
      const rawSs = SpreadsheetApp.openById(RAW_DATA_SPREADSHEET_ID);
      const rawSheet = rawSs.getSheetByName('Raw') || rawSs.getSheets()[0];
      
      // หาบรรทัดสุดท้ายที่มีข้อมูลจริงใน Column A
      const colAValues = rawSheet.getRange("A:A").getValues();
      let actualLastRow = 0;
      for (let i = colAValues.length - 1; i >= 0; i--) {
        if (colAValues[i][0] !== "" && colAValues[i][0] !== null && colAValues[i][0] !== undefined) {
          actualLastRow = i + 1;
          break;
        }
      }

      const rawLastCol = rawSheet.getLastColumn();

      if (actualLastRow >= 2) {
        const headers = rawSheet.getRange(1, 1, 1, rawLastCol).getValues()[0].map(h => h ? String(h).trim().toLowerCase() : '');
        
        const shipmentIdx = headers.indexOf('shipment_id');
        const reasonIdx = headers.indexOf('soc_outbound_late_type_2nd_cutoff');
        const hubIdx = headers.indexOf('dest_station_name');

        const rawValues = rawSheet.getRange(2, 1, actualLastRow - 1, rawLastCol).getValues();

        for (let i = 0; i < rawValues.length; i++) {
          let row = rawValues[i];
          
          let shipmentId = (shipmentIdx !== -1 && row[shipmentIdx]) ? String(row[shipmentIdx]).trim() : String(row[0] || '-').trim();
          
          let rawReason = '';
          if (reasonIdx !== -1 && row[reasonIdx] !== null && row[reasonIdx] !== undefined) {
            rawReason = String(row[reasonIdx]).trim().toLowerCase();
          } else {
            rawReason = String(row[8] || row[18] || '').trim().toLowerCase();
          }

          if (rawReason === 'undefined') rawReason = '';

          let hubName = (hubIdx !== -1 && row[hubIdx]) ? String(row[hubIdx]).trim() : String(row[9] || '-').trim();
          if (hubName === 'undefined') hubName = '-';

          let zone = String(row[19] || row[9] || 'N/A').trim().toUpperCase();
          if (zone === 'UNDEFINED' || !zone) zone = 'N/A';

          if (rawReason.indexOf('skip') !== -1) {
            totalSkipCases++;

            if (rawReason.indexOf('machine') !== -1) {
              machineCount++;
            } else if (rawReason.indexOf('system') !== -1) {
              systemCount++;
            }

            skipCases.push({
              rowNo: i + 2,
              shipmentId: shipmentId || '-',
              reason: rawReason || 'skip_outbound',
              zone: zone,
              hub: hubName
            });
            
            if (zone) {
              skipCountByZone[zone] = (skipCountByZone[zone] || 0) + 1;
            }
          }
        }
      }
    } catch (e) {
      Logger.log("Raw data read error: " + e.toString());
    }

    // 3. Volume Tracker Sheet
    let availableDates = [];
    let socActual = 0;
    let targetCol = -1;

    try {
      const volSs = SpreadsheetApp.openById(VOLUME_SPREADSHEET_ID);
      const volSheet = volSs.getSheetByName('SOC') || volSs.getSheets()[0];
      const volLastCol = volSheet.getLastColumn();

      if (volLastCol >= 16) { 
        const row1DisplayValues = volSheet.getRange(1, 1, 1, volLastCol).getDisplayValues()[0];
        const row10Values = volSheet.getRange(10, 1, 1, volLastCol).getValues()[0];

        for (let c = 15; c < row1DisplayValues.length; c++) {
          let dateStr = String(row1DisplayValues[c]).trim();
          if (dateStr !== "" && !dateStr.startsWith('#')) {
            availableDates.push({ colIndex: c, dateText: dateStr });
          }
        }

        targetCol = (selectedDateColIndex !== undefined && selectedDateColIndex !== null && selectedDateColIndex !== "")
          ? Number(selectedDateColIndex) 
          : (availableDates.length > 0 ? availableDates[0].colIndex : -1);

        if (targetCol >= 0 && row10Values[targetCol] !== undefined) {
          let val = Number(row10Values[targetCol]);
          if (!isNaN(val)) socActual = val;
        }
      }
    } catch (e) {
      Logger.log("Volume read error: " + e.toString());
    }

    let skipPercentage = socActual > 0 ? ((totalSkipCases / socActual) * 100).toFixed(2) : "0.00";

    return {
      staffByZone: staffByZone,
      skipCases: skipCases,
      skipCountByZone: skipCountByZone,
      totalSkipCases: totalSkipCases,
      machineCount: machineCount,
      systemCount: systemCount,
      socActual: socActual,
      skipPercentage: skipPercentage,
      availableDates: availableDates,
      selectedColIndex: targetCol
    };

  } catch (err) {
    return { error: err.toString() };
  }
}

// Clear ข้อมูลเก่าแบบถอนรากถอนโคน
function clearRawDataSheet() {
  try {
    const rawSs = SpreadsheetApp.openById(RAW_DATA_SPREADSHEET_ID);
    const rawSheet = rawSs.getSheetByName('Raw') || rawSs.getSheets()[0];
    
    const maxRows = rawSheet.getMaxRows();
    if (maxRows >= 2) {
      rawSheet.getRange(2, 1, maxRows - 1, 15).clearContent();
    }
    
    SpreadsheetApp.flush();
    return { success: true };
  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// ทยอยบันทึกต่อท้ายโดยเช็คบรรทัดที่มีข้อมูลจริงเท่านั้น
function appendCsvChunkToRawSheet(chunkData) {
  try {
    if (!chunkData || chunkData.length === 0) {
      return { success: true, count: 0 };
    }

    const rawSs = SpreadsheetApp.openById(RAW_DATA_SPREADSHEET_ID);
    const rawSheet = rawSs.getSheetByName('Raw') || rawSs.getSheets()[0];

    const colAValues = rawSheet.getRange("A:A").getValues();
    let actualLastRow = 0;
    for (let i = colAValues.length - 1; i >= 0; i--) {
      if (colAValues[i][0] !== "" && colAValues[i][0] !== null && colAValues[i][0] !== undefined) {
        actualLastRow = i + 1;
        break;
      }
    }

    const targetStartRow = Math.max(actualLastRow + 1, 2);

    const numRows = chunkData.length;
    const numCols = chunkData[0].length;

    rawSheet.getRange(targetStartRow, 1, numRows, numCols).setValues(chunkData);

    SpreadsheetApp.flush();
    return { success: true, count: numRows };

  } catch (err) {
    return { success: false, error: err.toString() };
  }
}

// Volume History Shared Server Storage (PropertiesService)
function getVolumeHistoryServer() {
  const props = PropertiesService.getScriptProperties();
  const raw = props.getProperty('VOLUME_DATA');
  if (raw) {
    try {
      const res = JSON.parse(raw);
      if (res && res.history && res.history.length > 0) return res;
    } catch(e) {}
  }
  const defaultData = {
    history: [
      { date: "2026-09-01", actual: 1814121 },
      { date: "2026-08-30", actual: 980457 }
    ],
    active: { date: "2026-08-30", actual: 980457 }
  };
  props.setProperty('VOLUME_DATA', JSON.stringify(defaultData));
  return defaultData;
}

function saveVolumeEntryServer(dateStr, actual, setActive) {
  const data = getVolumeHistoryServer();
  let history = data.history || [];
  const existingIdx = history.findIndex(h => h.date === dateStr);
  const entry = { date: dateStr, actual: parseInt(actual, 10) };
  if (existingIdx >= 0) history[existingIdx] = entry;
  else history.unshift(entry);
  history.sort((a, b) => b.date.localeCompare(a.date));
  data.history = history;
  if (setActive || !data.active) data.active = entry;
  PropertiesService.getScriptProperties().setProperty('VOLUME_DATA', JSON.stringify(data));
  return data;
}

function deleteVolumeEntryServer(dateStr) {
  const data = getVolumeHistoryServer();
  data.history = (data.history || []).filter(h => h.date !== dateStr);
  if (data.active && data.active.date === dateStr) {
    data.active = null;
  }
  PropertiesService.getScriptProperties().setProperty('VOLUME_DATA', JSON.stringify(data));
  return data;
}

function setActiveVolumeServer(dateStr, actual) {
  const data = getVolumeHistoryServer();
  if (!dateStr || !actual) {
    data.active = null;
  } else {
    data.active = { date: dateStr, actual: parseInt(actual, 10) };
  }
  PropertiesService.getScriptProperties().setProperty('VOLUME_DATA', JSON.stringify(data));
  return data;
}

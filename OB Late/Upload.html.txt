  <script>
    let targetMeta = null;   // { headers, currentRowCount, sheetName }
    let parsedHeaders = [];
    let parsedRows = [];

    function escapeHtml(s) {
      return String(s == null ? '' : s).replace(/[&<>"']/g, c => ({
        '&': '&amp;', '<': '&lt;', '>': '&gt;', '"': '&quot;', "'": '&#39;'
      }[c]));
    }

    function loadMeta() {
      google.script.run
        .withSuccessHandler(meta => {
          targetMeta = meta;
          document.getElementById('subtitle').textContent =
            'ปลายทาง: ชีต "' + meta.sheetName + '" · มีข้อมูลอยู่แล้ว ' + meta.currentRowCount.toLocaleString() + ' แถว';
        })
        .withFailureHandler(err => {
          document.getElementById('subtitle').textContent = 'โหลดข้อมูลปลายทางไม่สำเร็จ: ' + (err && err.message ? err.message : err);
        })
        .getRawDataMeta();
    }

    // ---- File selection ----------------------------------------------
    const dropzone = document.getElementById('dropzone');
    const fileInput = document.getElementById('fileInput');

    dropzone.addEventListener('click', () => fileInput.click());
    dropzone.addEventListener('dragover', e => { e.preventDefault(); dropzone.classList.add('drag-over'); });
    dropzone.addEventListener('dragleave', () => dropzone.classList.remove('drag-over'));
    dropzone.addEventListener('drop', e => {
      e.preventDefault();
      dropzone.classList.remove('drag-over');
      if (e.dataTransfer.files.length) handleFile(e.dataTransfer.files[0]);
    });
    fileInput.addEventListener('change', e => {
      if (e.target.files.length) handleFile(e.target.files[0]);
    });

    function handleFile(file) {
      if (!file.name.toLowerCase().endsWith('.csv')) {
        showValidation('error', 'กรุณาเลือกไฟล์ .csv เท่านั้น');
        return;
      }
      document.getElementById('fileChip').textContent = '📄 ' + file.name + ' (' + Math.round(file.size / 1024) + ' KB)';
      document.getElementById('fileChipWrap').style.display = 'block';

      Papa.parse(file, {
        header: false,
        skipEmptyLines: true,
        complete: results => {
          const data = results.data;
          if (!data.length) {
            showValidation('error', 'ไฟล์นี้ไม่มีข้อมูล');
            return;
          }
          parsedHeaders = data[0];
          parsedRows = data.slice(1);
          onParsed();
        },
        error: err => showValidation('error', 'อ่านไฟล์ไม่สำเร็จ: ' + err.message)
      });
    }

    function onParsed() {
      validateHeaders();
      renderPreview();
      document.getElementById('previewCard').style.display = 'block';
      document.getElementById('resultBox').style.display = 'none';
    }

    function validateHeaders() {
      if (!targetMeta || !targetMeta.headers || !targetMeta.headers.length) {
        showValidation('warn', 'ยังไม่ทราบโครงสร้างคอลัมน์ปลายทาง จะจับคู่คอลัมน์ตามชื่อหัวข้อตอนอัปโหลด');
        return;
      }
      const targetSet = new Set(targetMeta.headers);
      const matched = parsedHeaders.filter(h => targetSet.has(h));
      const unmatched = parsedHeaders.filter(h => !targetSet.has(h));

      if (matched.length === parsedHeaders.length) {
        showValidation('ok', '✓ หัวข้อคอลัมน์ตรงกับ Sheet ปลายทางครบทุกคอลัมน์ (' + matched.length + ' คอลัมน์)');
      } else if (matched.length > 0) {
        showValidation('warn', '⚠ จับคู่ได้ ' + matched.length + '/' + parsedHeaders.length + ' คอลัมน์ — คอลัมน์ที่ไม่ตรง: ' + unmatched.join(', '));
      } else {
        showValidation('error', '✗ ไม่พบคอลัมน์ที่ตรงกับ Sheet ปลายทางเลย ตรวจสอบไฟล์อีกครั้ง');
      }
    }

    function showValidation(level, text) {
      const box = document.getElementById('validationBox');
      box.innerHTML = '<div class="validation ' + level + '">' + escapeHtml(text) + '</div>';
    }

    function renderPreview() {
      document.getElementById('statRows').textContent = parsedRows.length.toLocaleString() + ' แถวในไฟล์';
      document.getElementById('statCols').textContent = parsedHeaders.length + ' คอลัมน์';
      document.getElementById('statExisting').textContent =
        'มีอยู่แล้ว ' + (targetMeta ? targetMeta.currentRowCount.toLocaleString() : '?') + ' แถวใน Sheet';

      const previewRows = parsedRows.slice(0, 5);
      let html = '<thead><tr>' + parsedHeaders.map(h => '<th>' + escapeHtml(h) + '</th>').join('') + '</tr></thead><tbody>';
      previewRows.forEach(row => {
        html += '<tr>' + row.map(c => '<td>' + escapeHtml(c) + '</td>').join('') + '</tr>';
      });
      html += '</tbody>';
      document.getElementById('previewTable').innerHTML = html;
    }

    document.getElementById('clearBtn').addEventListener('click', () => {
      parsedHeaders = [];
      parsedRows = [];
      fileInput.value = '';
      document.getElementById('fileChipWrap').style.display = 'none';
      document.getElementById('previewCard').style.display = 'none';
      document.getElementById('validationBox').innerHTML = '';
    });

    document.getElementById('uploadBtn').addEventListener('click', () => {
      if (!parsedRows.length) return;

      const mode = document.querySelector('input[name="mode"]:checked').value;
      const skipDuplicates = mode === 'append' && document.getElementById('skipDupes').checked;

      document.getElementById('uploadBtn').disabled = true;
      document.getElementById('truckLoader').style.display = 'block';
      document.getElementById('resultBox').style.display = 'none';

      google.script.run
        .withSuccessHandler(onUploadSuccess)
        .withFailureHandler(onUploadError)
        .uploadRawData({
          headers: parsedHeaders,
          rows: parsedRows,
          mode: mode,
          skipDuplicates: skipDuplicates
        });
    });

    function onUploadSuccess(result) {
      document.getElementById('truckLoader').style.display = 'none';
      document.getElementById('uploadBtn').disabled = false;

      const box = document.getElementById('resultBox');
      box.className = 'result-box ok';
      box.style.display = 'block';
      box.textContent = '✓ อัปโหลดสำเร็จ: เขียน ' + result.written.toLocaleString() + ' แถว' +
        (result.skipped ? ' (ข้าม ' + result.skipped.toLocaleString() + ' แถวซ้ำ)' : '') +
        ' — ตอนนี้ Sheet มีทั้งหมด ' + result.totalAfter.toLocaleString() + ' แถว';

      loadMeta();
    }

    function onUploadError(err) {
      document.getElementById('truckLoader').style.display = 'none';
      document.getElementById('uploadBtn').disabled = false;

      const box = document.getElementById('resultBox');
      box.className = 'result-box error';
      box.style.display = 'block';
      box.textContent = '✗ อัปโหลดไม่สำเร็จ: ' + (err && err.message ? err.message : err);
    }

    loadMeta();
  </script>

// Folder: https://drive.google.com/drive/folders/1AwxaJv04MQ4g1l3MSOCSsC5bQrAaXoI1
const CHARTS_EXPORT_FOLDER_ID = '1AwxaJv04MQ4g1l3MSOCSsC5bQrAaXoI1';

function exportChartsToSheet(chartsData, fileName) {
  const name = (fileName && String(fileName).trim())
    ? String(fileName).trim()
    : 'Dashboard Charts - ' + Utilities.formatDate(new Date(), Session.getScriptTimeZone() || 'GMT', 'yyyy-MM-dd HH:mm');

  // --- สร้างไฟล์ Google Sheet ใหม่ทั้งไฟล์ ---
  const newSs = SpreadsheetApp.create(name);
  const sheet = newSs.getSheets()[0];
  sheet.setName('Dashboard Charts');

  // --- ย้ายไฟล์ไปไว้ใน Drive folder ที่กำหนด ---
  const file = DriveApp.getFileById(newSs.getId());
  const targetFolder = DriveApp.getFolderById(CHARTS_EXPORT_FOLDER_ID);
  targetFolder.addFile(file);
  // เอาไฟล์ออกจาก root ของ My Drive กันไม่ให้ไปโผล่ซ้ำที่นั่นด้วย
  const rootFolder = DriveApp.getRootFolder();
  try { rootFolder.removeFile(file); } catch (e) { /* อาจไม่ได้อยู่ที่ root ก็ไม่เป็นไร */ }

  // ลำดับกราฟที่จะ export — ต้องตรงกับ key ที่ส่งมาจาก window.chartExportData
  const CHART_ORDER = ['pieStatus', 'pieVehicle', 'pieOnTime', 'barCut', 'barHour'];

  const DATA_COL_START = 1;   // คอลัมน์ A: ตารางข้อมูลดิบของกราฟ
  const CHART_ANCHOR_COL = 6; // คอลัมน์ F: ตำแหน่งวางกราฟ
  const ROW_SLOT_HEIGHT = 18; // จำนวนแถวที่เผื่อไว้ต่อกราฟ 1 ตัว (กันซ้อนทับ)

  let dataRow = 1;
  let chartCount = 0;

  CHART_ORDER.forEach((key, idx) => {
    const chart = chartsData[key];
    if (!chart || !chart.labels || !chart.labels.length) return;

    const seriesNames = Object.keys(chart.series);

    // --- หัวข้อกราฟ ---
    sheet.getRange(dataRow, DATA_COL_START).setValue(chart.title).setFontWeight('bold');
    dataRow++;

    // --- แถวหัวตาราง: Label, Series1, Series2, ... ---
    const headerRow = ['Label'].concat(seriesNames);
    sheet.getRange(dataRow, DATA_COL_START, 1, headerRow.length)
      .setValues([headerRow])
      .setFontWeight('bold');
    const headerRowIdx = dataRow;
    dataRow++;

    // --- แถวข้อมูล ---
    const dataRows = chart.labels.map((label, i) => {
      return [label].concat(seriesNames.map(s => chart.series[s][i] || 0));
    });
    sheet.getRange(dataRow, DATA_COL_START, dataRows.length, headerRow.length).setValues(dataRows);
    dataRow += dataRows.length + 2; // เว้นบรรทัดว่างก่อนตารางถัดไป

    // --- สร้างกราฟจริงของ Google Sheets จากตารางข้อมูลด้านบน ---
    const dataRange = sheet.getRange(headerRowIdx, DATA_COL_START, dataRows.length + 1, headerRow.length);
    const anchorRow = 1 + idx * ROW_SLOT_HEIGHT;

    let builder = sheet.newChart()
      .addRange(dataRange)
      .setPosition(anchorRow, CHART_ANCHOR_COL, 0, 0)
      .setOption('title', chart.title)
      .setOption('width', 520)
      .setOption('height', 320);

    if (chart.type === 'PIE') {
      builder = builder.setChartType(Charts.ChartType.PIE).setOption('pieSliceText', 'percentage');
    } else {
      builder = builder
        .setChartType(Charts.ChartType.COLUMN)
        .setOption('isStacked', seriesNames.length > 1);
    }

    sheet.insertChart(builder.build());
    chartCount++;
  });

  SpreadsheetApp.flush();

  return {
    url: newSs.getUrl(),
    fileName: name,
    chartCount: chartCount
  };
}

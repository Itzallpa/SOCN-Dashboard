    function escapeHtml(str) {
      return String(str == null ? '' : str)
        .replace(/&/g, '&amp;')
        .replace(/</g, '&lt;')
        .replace(/>/g, '&gt;')
        .replace(/"/g, '&quot;');
    }

    // Build the headers used for display/export, inserting a standalone
    // "Late Cut" column right after "Status" instead of leaving that
    // info folded into the Status column.
    function getDisplayHeaders() {
      const headers = (window.__dashboardHeaders || []).slice();
      headers.splice(COL.STATUS + 1, 0, 'Late Cut');
      return headers;
    }

    // Build a row's cells to match getDisplayHeaders(), inserting the
    // computed "Late Cut" value (only populated for late trips).
    function getDisplayCells(cells) {
      const out = cells.slice();
      const isLate = String(cells[COL.STATUS] || '').trim().toLowerCase() === 'late';
      const lateCut = isLate ? (applicableCut(cells) || '') : '';
      out.splice(COL.STATUS + 1, 0, lateCut);
      return out;
    }

    // ---- Raw-data modal ----
    function showRawDataModal(title, subsetRows) {
      currentModalTitle = title;
      currentModalSubset = subsetRows;

      const headers = getDisplayHeaders();
      document.getElementById('modalTitle').textContent = title;
      document.getElementById('modalMeta').textContent =
        subsetRows.length.toLocaleString() + ' เที่ยว (raw data ที่ใช้คำนวณกราฟนี้)';

      let html = '<thead><tr>' + headers.map(h => '<th>' + escapeHtml(h) + '</th>').join('') + '</tr></thead><tbody>';
      subsetRows.forEach(r => {
        html += '<tr>' + getDisplayCells(r.cells).map(c => '<td>' + escapeHtml(c) + '</td>').join('') + '</tr>';
      });
      html += '</tbody>';
      document.getElementById('modalTable').innerHTML = html;
      document.getElementById('modalOverlay').classList.add('open');
    }

    function closeRawDataModal() {
      document.getElementById('modalOverlay').classList.remove('open');
    }

    // Export raw data subset in Modal to CSV
    function exportModalToCSV() {
      if (!currentModalSubset || !currentModalSubset.length) return;

      const headers = getDisplayHeaders();
      const rows = [headers, ...currentModalSubset.map(r => getDisplayCells(r.cells))];

      const csvContent = '\uFEFF' + rows.map(e => e.map(val => {
        const str = String(val == null ? '' : val).replace(/"/g, '""');
        return `"${str}"`;
      }).join(',')).join('\r\n');

      const blob = new Blob([csvContent], { type: 'text/csv;charset=utf-8;' });
      const url = URL.createObjectURL(blob);
      const link = document.createElement('a');
      const safeTitle = (currentModalTitle || 'raw_data').replace(/[^a-zA-Z0-9ก-๙_-]/g, '_');

      link.setAttribute('href', url);
      link.setAttribute('download', `${safeTitle}.csv`);
      document.body.appendChild(link);
      link.click();
      document.body.removeChild(link);
    }

    function load(forceRefresh) {
      document.getElementById('stateMsg').style.display = 'block';
      document.getElementById('content').style.display = 'none';
      document.getElementById('errorBox').style.display = 'none';

      const runner = google.script.run
        .withSuccessHandler(onData)
        .withFailureHandler(onError);

      if (forceRefresh) {
        runner.forceRefreshTableData();
      } else {
        runner.getTableData(false);
      }
    }

    function loadOnTimeArrival(forceRefresh) {
      const runner = google.script.run
        .withSuccessHandler(onOnTimeArrivalData)
        .withFailureHandler((err) => console.error('on time arrival load failed:', err));
      if (forceRefresh) runner.forceRefreshOnTimeArrivalData();
      else runner.getOnTimeArrivalData(false);
    }

    function onOnTimeArrivalData(data) {
      onTimeArrivalMap = (data && data.map) || {};
      if (allRows.length) updateActiveRows();
    }

    function getOnTimeArrivalCategory(row) {
      const trip = (row.cells[COL.TRIP_NO] || '').trim();
      if (!trip || !(trip in onTimeArrivalMap)) return null;
      
      const otaStatus = (onTimeArrivalMap[trip] || '').trim().toLowerCase();
      const tableStatus = (row.cells[COL.STATUS] || '').trim().toLowerCase();
      
      const isOtaPending = otaStatus.indexOf('pending') !== -1;
      const isOtaOnTime = otaStatus === 'on-time' || otaStatus === 'on time' || otaStatus.indexOf('on-time') !== -1 || otaStatus.indexOf('on time') !== -1;
      const isOtaLate = otaStatus === 'late' || otaStatus.indexOf('late') !== -1;
      
      const isTableOnTime = tableStatus === 'on-time' || tableStatus === 'on time';
      const isTableLate = tableStatus === 'late';
      
      if (isOtaPending) {
        return 'RC';
      }
      if (isOtaOnTime && isTableOnTime) {
        return 'On time';
      }
      if (isOtaLate && isTableLate) {
        return 'LH Late';
      }
      if (isOtaOnTime && isTableLate) {
        return 'OB Late';
      }
      return null;
    }

    function onError(err) {
      document.getElementById('stateMsg').style.display = 'none';
      const box = document.getElementById('errorBox');
      box.style.display = 'block';
      box.textContent = 'Could not load data: ' + (err && err.message ? err.message : err);
    }

    function applicableCut(cells) {
      for (const c of CUT_COLS) {
        const v = (cells[c.idx] || '').trim();
        if (v && v !== '-') return c.label;
      }
      return null;
    }

    function hourOf(timeStr) {
      if (!timeStr) return null;
      const s = String(timeStr).trim();
      const m = s.match(/(\d{1,2}):(\d{2})(?::\d{2})?\s*([AaPp][Mm])?/);
      if (!m) return null;
      let h = parseInt(m[1], 10);
      const ampm = m[3] ? m[3].toUpperCase() : null;
      if (ampm === 'PM' && h < 12) h += 12;
      if (ampm === 'AM' && h === 12) h = 0;
      if (isNaN(h) || h < 0 || h > 23) return null;
      return h;
    }

    function destroyChart(name) {
      if (charts[name]) {
        charts[name].destroy();
        charts[name] = null;
      }
    }

    function filterRowsByCut(rows, cutName) {
      if (!cutName || cutName === 'All Cuts') return rows.slice();
      return rows.filter(r => applicableCut(r.cells) === cutName);
    }

    function uniqueRegions(rows) {
      const seen = [];
      rows.forEach(r => {
        const region = (r.cells[COL.REGION] || '').trim();
        if (region && seen.indexOf(region) === -1) {
          seen.push(region);
        }
      });
      return seen.sort();
    }

    function filterRowsByRegion(rows, regionName) {
      if (!regionName || regionName === 'All Regions') return rows.slice();
      return rows.filter(r => (r.cells[COL.REGION] || '').trim() === regionName);
    }

    function uniqueZones(rows) {
      const seen = [];
      rows.forEach(r => {
        const zone = (r.cells[COL.ZONE] || '').trim();
        if (zone && seen.indexOf(zone) === -1) {
          seen.push(zone);
        }
      });
      return seen.sort();
    }

    function filterRowsByZone(rows, zoneName) {
      if (!zoneName || zoneName === 'All Zones') return rows.slice();
      return rows.filter(r => (r.cells[COL.ZONE] || '').trim() === zoneName);
    }

    function getBaseFilteredRows() {
      return allRows.filter(r => {
        if (activeIntentType === 'Intentional') {
          const val = (r.cells[COL.INTENTIONAL] || '').trim();
          if (!val || val === '-') return false;
        } else if (activeIntentType === 'Non Intentional') {
          const val = (r.cells[COL.NON_INTENTIONAL] || '').trim();
          if (!val || val === '-') return false;
        }
        return true;
      });
    }

    function updateActiveRows() {
      const baseRows = getBaseFilteredRows();
      const regionFiltered = filterRowsByRegion(baseRows, activeRegion);
      const zoneFiltered = filterRowsByZone(regionFiltered, activeZone);
      activeRows = filterRowsByCut(zoneFiltered, activeCut);
      renderDashboard(activeRows);
    }

    function setActiveZone(zoneName) {
      activeZone = zoneName || 'All Zones';
      const zoneSelect = document.getElementById('zoneSelect');
      if (zoneSelect && zoneSelect.value !== activeZone) {
        zoneSelect.value = activeZone;
      }
      updateActiveRows();
    }

    function setActiveCut(cutName) {
      activeCut = cutName || 'All Cuts';
      const cutSelect = document.getElementById('cutSelect');
      if (cutSelect && cutSelect.value !== activeCut) {
        cutSelect.value = activeCut;
      }
      updateActiveRows();
    }

    function setActiveRegion(regionName) {
      activeRegion = regionName || 'All Regions';
      const regionSelect = document.getElementById('regionSelect');
      if (regionSelect && regionSelect.value !== activeRegion) {
        regionSelect.value = activeRegion;
      }
      updateActiveRows();
    }

    function renderDashboard(rows) {
      const onTimeRows = rows.filter(r => String(r.cells[COL.STATUS] || '').trim().toLowerCase() === 'on time');
      const lateRows = rows.filter(r => String(r.cells[COL.STATUS] || '').trim().toLowerCase() === 'late');
      const withStatus = onTimeRows.length + lateRows.length;
      const rate = withStatus ? Math.round((onTimeRows.length / withStatus) * 1000) / 10 : 0;

      document.getElementById('subtitle').textContent =
        rows.length + ' trips · updated ' + new Date(window.__dashboardGeneratedAt || Date.now()).toLocaleString() +
        (activeIntentType === 'All' ? '' : ' · ' + activeIntentType) +
        (activeRegion === 'All Regions' ? '' : ' · ' + activeRegion) +
        (activeZone === 'All Zones' ? '' : ' · ' + activeZone) +
        (activeCut === 'All Cuts' ? '' : ' · ' + activeCut);

      document.getElementById('kpiTotal').textContent = rows.length.toLocaleString();
      document.getElementById('kpiTotalSub').textContent = withStatus + ' with a known status';
      document.getElementById('kpiOnTime').textContent = onTimeRows.length.toLocaleString();
      document.getElementById('kpiOnTimeSub').textContent = withStatus ? Math.round((onTimeRows.length / withStatus) * 100) + '% of trips' : '';
      document.getElementById('kpiLate').textContent = lateRows.length.toLocaleString();
      document.getElementById('kpiLateSub').textContent = withStatus ? Math.round((lateRows.length / withStatus) * 100) + '% of trips' : '';
      document.getElementById('kpiRate').textContent = rate + '%';

      const hourCounts = {};
      lateRows.forEach(r => {
        const h = hourOf(r.cells[COL.ACTUAL_DEP_CUT]);
        if (h === null) return;
        hourCounts[h] = (hourCounts[h] || 0) + 1;
      });
      const hourEntries = Object.entries(hourCounts).sort((a, b) => b[1] - a[1]);
      if (hourEntries.length) {
        const [peakH, peakCount] = hourEntries[0];
        const hh = String(peakH).padStart(2, '0');
        const hhNext = String((parseInt(peakH, 10) + 1) % 24).padStart(2, '0');
        document.getElementById('peakHour').textContent = hh + ':00 – ' + hhNext + ':00';
        document.getElementById('peakHourSub').textContent = peakCount + ' เที่ยวที่สายในช่วงนี้ (จาก ' + lateRows.length + ' เที่ยวที่สายทั้งหมด)';
      } else {
        document.getElementById('peakHour').textContent = 'ไม่มีข้อมูล';
        document.getElementById('peakHourSub').textContent = '';
      }

      const baseRowsForCut = getBaseFilteredRows();
      const cutLateCounts = {};
      const cutTotalCounts = {};
      baseRowsForCut.forEach(r => {
        const cut = applicableCut(r.cells);
        if (!cut) return;
        cutTotalCounts[cut] = (cutTotalCounts[cut] || 0) + 1;
        if (String(r.cells[COL.STATUS] || '').trim().toLowerCase() === 'late') {
          cutLateCounts[cut] = (cutLateCounts[cut] || 0) + 1;
        }
      });

      const cutSelect = document.getElementById('cutSelect');
      const regionSelect = document.getElementById('regionSelect');
      const zoneSelect = document.getElementById('zoneSelect');
      const availableCuts = CUT_COLS.map(c => c.label).filter(l => cutTotalCounts[l]);
      const availableRegions = uniqueRegions(baseRowsForCut);
      const availableZones = uniqueZones(baseRowsForCut);

      regionSelect.innerHTML = '<option value="All Regions">All Regions</option>' +
        availableRegions.map(r => '<option value="' + escapeHtml(r) + '">' + escapeHtml(r) + '</option>').join('');
      regionSelect.value = activeRegion;

      zoneSelect.innerHTML = '<option value="All Zones">All Zones</option>' +
        availableZones.map(z => '<option value="' + escapeHtml(z) + '">' + escapeHtml(z) + '</option>').join('');
      zoneSelect.value = activeZone;

      cutSelect.innerHTML = '<option value="All Cuts">All Cuts</option>' +
        availableCuts.map(l => '<option value="' + l + '">' + l + '</option>').join('');
      cutSelect.value = activeCut;

      const cutName = activeCut === 'All Cuts' ? (Object.entries(cutLateCounts).sort((a, b) => b[1] - a[1])[0]?.[0] || 'ไม่มีข้อมูล') : activeCut;
      const total = cutTotalCounts[cutName] || 0;
      const late = cutLateCounts[cutName] || 0;
      const pct = total ? Math.round((late / total) * 100) : 0;
      document.getElementById('peakCut').textContent = cutName;
      document.getElementById('peakCutSub').textContent = total ? late + ' จาก ' + total + ' เที่ยว (' + pct + '% สาย)' : '';

      destroyChart('pieStatus');
      destroyChart('pieVehicle');
      destroyChart('pieOnTime');
      destroyChart('barCut');
      destroyChart('barHour');

      charts.pieStatus = new Chart(document.getElementById('pieStatus'), {
        type: 'pie',
        data: {
          labels: ['On Time', 'Late'],
          datasets: [{
            data: [onTimeRows.length, lateRows.length],
            backgroundColor: ['#0f9d58', '#e0293e'],
            borderWidth: 0
          }]
        },
        options: {
          maintainAspectRatio: false,
          onClick: (evt, elements, chart) => {
            if (!elements.length) return;
            const label = chart.data.labels[elements[0].index];
            if (label === 'On Time') showRawDataModal((activeCut === 'All Cuts' ? '' : activeCut + ' · ') + 'On Time — raw data', onTimeRows);
            else if (label === 'Late') showRawDataModal((activeCut === 'All Cuts' ? '' : activeCut + ' · ') + 'Late — raw data', lateRows);
          },
          onHover: (evt, elements) => {
            evt.native.target.style.cursor = elements.length ? 'pointer' : 'default';
          },
          plugins: {
            legend: {
              position: 'bottom',
              labels: {
                generateLabels: (chart) => {
                  const ds = chart.data.datasets[0];
                  const total = ds.data.reduce((a, b) => a + b, 0);
                  return chart.data.labels.map((label, i) => {
                    const val = ds.data[i];
                    const pct = total ? Math.round((val / total) * 1000) / 10 : 0;
                    return {
                      text: label + ' — ' + val.toLocaleString() + ' (' + pct + '%)',
                      fillStyle: ds.backgroundColor[i],
                      index: i
                    };
                  });
                }
              }
            },
            tooltip: {
              callbacks: {
                label: (ctx) => {
                  const total = ctx.dataset.data.reduce((a, b) => a + b, 0);
                  const pct = total ? Math.round((ctx.parsed / total) * 1000) / 10 : 0;
                  return ctx.label + ': ' + ctx.parsed.toLocaleString() + ' (' + pct + '%)';
                }
              }
            },
            datalabels: {
              display: true,
              color: '#fff',
              font: { weight: 800, size: 12 },
              formatter: (value, ctx) => {
                const total = ctx.dataset.data.reduce((a, b) => a + b, 0);
                if (!total || !value) return '';
                return Math.round((value / total) * 1000) / 10 + '%';
              }
            }
          }
        }
      });

      const vehLateCounts = {};
      lateRows.forEach(r => {
        const v = r.cells[COL.VEHICLE_TYPE] || 'Unknown';
        vehLateCounts[v] = (vehLateCounts[v] || 0) + 1;
      });
      const vehEntries = Object.entries(vehLateCounts).sort((a, b) => b[1] - a[1]);
      charts.pieVehicle = new Chart(document.getElementById('pieVehicle'), {
        type: 'pie',
        data: {
          labels: vehEntries.map(e => e[0]),
          datasets: [{
            data: vehEntries.map(e => e[1]),
            backgroundColor: vehEntries.map((_, i) => PALETTE[i % PALETTE.length]),
            borderWidth: 0
          }]
        },
        options: {
          maintainAspectRatio: false,
          onClick: (evt, elements, chart) => {
            if (!elements.length) return;
            const label = chart.data.labels[elements[0].index];
            const subset = lateRows.filter(r => (r.cells[COL.VEHICLE_TYPE] || 'Unknown') === label);
            showRawDataModal((activeCut === 'All Cuts' ? '' : activeCut + ' · ') + 'เที่ยวที่สาย — ' + label, subset);
          },
          onHover: (evt, elements) => {
            evt.native.target.style.cursor = elements.length ? 'pointer' : 'default';
          },
          plugins: {
            legend: {
              position: 'bottom',
              labels: {
                boxWidth: 12,
                font: { size: 11 },
                generateLabels: (chart) => {
                  const ds = chart.data.datasets[0];
                  const total = ds.data.reduce((a, b) => a + b, 0);
                  return chart.data.labels.map((label, i) => {
                    const val = ds.data[i];
                    const pct = total ? Math.round((val / total) * 1000) / 10 : 0;
                    return {
                      text: label + ' — ' + val.toLocaleString() + ' (' + pct + '%)',
                      fillStyle: ds.backgroundColor[i],
                      index: i
                    };
                  });
                }
              }
            },
            tooltip: {
              callbacks: {
                label: (ctx) => {
                  const total = ctx.dataset.data.reduce((a, b) => a + b, 0);
                  const pct = total ? Math.round((ctx.parsed / total) * 1000) / 10 : 0;
                  return ctx.label + ': ' + ctx.parsed.toLocaleString() + ' (' + pct + '%)';
                }
              }
            },
            datalabels: {
              display: true,
              color: '#fff',
              font: { weight: 800, size: 10.5 },
              formatter: (value, ctx) => {
                const total = ctx.dataset.data.reduce((a, b) => a + b, 0);
                if (!total || !value) return '';
                const pct = Math.round((value / total) * 1000) / 10;
                return pct >= 4 ? pct + '%' : '';
              }
            }
          }
        }
      });

      const OTA_CATS = ['On time', 'LH Late', 'OB Late', 'RC'];
      const OTA_COLORS = { 'On time': '#0f9d58', 'LH Late': '#e0293e', 'OB Late': '#f5a623', 'RC': '#8a8f98' };
      const otaRowsByCat = { 'On time': [], 'LH Late': [], 'OB Late': [], 'RC': [] };
      rows.forEach(r => {
        const cat = getOnTimeArrivalCategory(r);
        if (cat && otaRowsByCat[cat]) otaRowsByCat[cat].push(r);
      });
      const otaLabels = OTA_CATS.filter(c => otaRowsByCat[c].length > 0);
      charts.pieOnTime = new Chart(document.getElementById('pieOnTime'), {
        type: 'pie',
        data: {
          labels: otaLabels,
          datasets: [{
            data: otaLabels.map(l => otaRowsByCat[l].length),
            backgroundColor: otaLabels.map(l => OTA_COLORS[l]),
            borderWidth: 0
          }]
        },
        options: {
          maintainAspectRatio: false,
          onClick: (evt, elements, chart) => {
            if (!elements.length) return;
            const label = chart.data.labels[elements[0].index];
            showRawDataModal((activeCut === 'All Cuts' ? '' : activeCut + ' · ') + label + ' — raw data', otaRowsByCat[label]);

    function renderHubTable(tbodyId, top20Hubs) {
      const hubTableBody = document.getElementById(tbodyId);
      if (!hubTableBody) return;
      if (top20Hubs.length === 0) {
        hubTableBody.innerHTML = '<tr><td colspan="6" style="text-align:center; padding:20px; color:var(--ink-soft);">ไม่มีข้อมูลการสาย (No late data)</td></tr>';
        return;
      }
      hubTableBody.innerHTML = top20Hubs.map((h, idx) => {
        const pct = h.totalCount ? ((h.lateCount / h.totalCount) * 100).toFixed(1) : '0.0';

        const cutBadges = [];
        if (h.cutLateCounts['Cut 0'] > 0) cutBadges.push(`<span class="cut-badge cut0">Cut 0: ${h.cutLateCounts['Cut 0']}</span>`);
        if (h.cutLateCounts['Cut 1'] > 0) cutBadges.push(`<span class="cut-badge cut1">Cut 1: ${h.cutLateCounts['Cut 1']}</span>`);
        if (h.cutLateCounts['Cut 2'] > 0) cutBadges.push(`<span class="cut-badge cut2">Cut 2: ${h.cutLateCounts['Cut 2']}</span>`);
        if (h.cutLateCounts['Cut 3'] > 0) cutBadges.push(`<span class="cut-badge cut3">Cut 3: ${h.cutLateCounts['Cut 3']}</span>`);
        if (h.cutLateCounts['Unknown'] > 0) cutBadges.push(`<span class="cut-badge cut-unknown">Unknown: ${h.cutLateCounts['Unknown']}</span>`);

        return `
          <tr>
            <td style="padding:10px; font-weight:700;">${idx + 1}</td>
            <td style="padding:10px; font-weight:700; color:var(--accent);">${escapeHtml(h.hub)}</td>
            <td style="padding:10px;">${escapeHtml(h.zone)}</td>
            <td style="padding:10px; text-align:center; font-weight:700; color:var(--red);">${h.lateCount} <span style="font-weight:normal; color:var(--ink-soft); font-size:11px;">/ ${h.totalCount}</span></td>
            <td style="padding:10px; text-align:center; font-weight:700;">${pct}%</td>
            <td style="padding:10px;">${cutBadges.join(' ')}</td>
          </tr>
        `;
      }).join('');
    }

    // Builds the Top 20 hubs list for a given On Time Arrival category
    // ('LH Late' or 'OB Late'), using getOnTimeArrivalCategory() so this
    // splits cleanly instead of lumping both late types into one table.
    function getTop20HubsByCategory(rows, category) {
      const hubData = {};
      rows.forEach(r => {
        const hub = (r.cells[COL.DEST] || '').trim();
        if (!hub || hub === '-') return;

        const zone = (r.cells[COL.ZONE] || 'Unknown').trim();
        const cat = getOnTimeArrivalCategory(r);
        const cut = applicableCut(r.cells) || 'Unknown';

        if (!hubData[hub]) {
          hubData[hub] = {
            hub: hub,
            zone: zone,
            totalCount: 0,
            lateCount: 0,
            cutLateCounts: { 'Cut 0': 0, 'Cut 1': 0, 'Cut 2': 0, 'Cut 3': 0, 'Unknown': 0 }
          };
        }

        hubData[hub].totalCount++;
        if (cat === category) {
          hubData[hub].lateCount++;
          if (hubData[hub].cutLateCounts[cut] !== undefined) {
            hubData[hub].cutLateCounts[cut]++;
          } else {
            hubData[hub].cutLateCounts[cut] = 1;
          }
        }
      });

      const hubList = Object.values(hubData).filter(h => h.lateCount > 0);
      hubList.sort((a, b) => {
        if (b.lateCount !== a.lateCount) return b.lateCount - a.lateCount;
        return b.totalCount - a.totalCount;
      });

      return hubList.slice(0, 20);
    }

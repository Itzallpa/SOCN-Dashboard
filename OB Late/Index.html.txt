  <script>
    // ---- State --------------------------------------------------------
    let HEADERS = [];
    let ROWS = [];
    let FILTERED = [];
    let sortCol = null;
    let sortDir = 1;
    let onTimeArrivalMap = {};

    const COL = {
      NO: 0, TRIP_NO: 1, TRIP_CATEGORY: 2, VEHICLE_TYPE: 3, PLATE: 4,
      DRIVER: 5, ORIGIN: 6, DEST: 7,
      CUT0: 8, CUT1: 9, CUT2: 10, CUT3: 11, ACTUAL_DEP_CUT: 12,
      DAY: 13, STATUS: 14, RMK: 15, NON_INTENTIONAL: 16, INTENTIONAL: 17,
      REGION: 18, ZONE: 19, ROUTE: 20
    };

    const searchable = [COL.TRIP_NO, COL.DRIVER, COL.PLATE, COL.ORIGIN, COL.DEST, COL.VEHICLE_TYPE];
    const chipColClasses = ['c0','c1','c2','c3','c4','c5'];
    const categoryColorMap = {};

    function colorForCategory(val) {
      if (!(val in categoryColorMap)) {
        const idx = Object.keys(categoryColorMap).length % chipColClasses.length;
        categoryColorMap[val] = chipColClasses[idx];
      }
      return categoryColorMap[val];
    }

    // ---- Late Cut Logic -----------------------------------------------
    function getLateCutDetail(row) {
      const status = (row.cells[COL.STATUS] || '').trim().toLowerCase();
      if (status !== 'late') return null;

      const actualDep = (row.cells[COL.ACTUAL_DEP_CUT] || '').trim();
      const cuts = [
        { name: 'Cut 0', val: (row.cells[COL.CUT0] || '').trim() },
        { name: 'Cut 1', val: (row.cells[COL.CUT1] || '').trim() },
        { name: 'Cut 2', val: (row.cells[COL.CUT2] || '').trim() },
        { name: 'Cut 3', val: (row.cells[COL.CUT3] || '').trim() }
      ];

      if (actualDep) {
        for (let i = cuts.length - 1; i >= 0; i--) {
          if (cuts[i].val && actualDep > cuts[i].val) {
            return cuts[i].name;
          }
        }
      }

      for (let i = cuts.length - 1; i >= 0; i--) {
        if (cuts[i].val) return cuts[i].name;
      }

      return null;
    }

    // ---- OTA Matching -------------------------------------------------
    function getOTACategory(row) {
      const trip = (row.cells[COL.TRIP_NO] || '').trim();
      if (!trip || !(trip in onTimeArrivalMap)) return null;
      const ota = (onTimeArrivalMap[trip] || '').trim().toLowerCase();
      const tbl = (row.cells[COL.STATUS] || '').trim().toLowerCase();
      const otaOnTime = ota === 'on-time' || ota === 'on time';
      const otaLate = ota === 'late';
      const otaPending = ota === 'pending';
      const tblOnTime = tbl === 'on-time' || tbl === 'on time';
      const tblLate = tbl === 'late';
      if (otaPending) return 'RC';
      if (otaOnTime && tblOnTime) return 'On time';
      if (otaLate && tblLate) return 'LH Late';
      if (otaOnTime && tblLate) return 'OB Late';
      return null;
    }

    function otaBadgeHtml(cat) {
      if (!cat) return '<span class="badge ota-none">—</span>';
      const cls = cat === 'RC' ? 'ota-rc' : cat === 'On time' ? 'ota-ontime' : cat === 'LH Late' ? 'ota-lhlate' : 'ota-oblate';
      return '<span class="badge ' + cls + '">' + escapeHtml(cat) + '</span>';
    }

    // ---- Boot ---------------------------------------------------------
    function load(forceRefresh) {
      document.getElementById('stateMsg').style.display = 'block';
      document.getElementById('stateMsg').innerHTML = truckLoaderHtml(
        forceRefresh ? 'Force-refreshing live from Google Sheets…' : 'Fetching rows from Google Sheets…'
      );
      document.getElementById('dataTable').style.display = 'none';
      document.getElementById('errorBox').style.display = 'none';
      const runner = google.script.run.withSuccessHandler(onData).withFailureHandler(onError);
      if (forceRefresh) runner.forceRefreshTableData(); else runner.getTableData(false);
    }

    function loadOnTimeArrival(forceRefresh) {
      const runner = google.script.run
        .withSuccessHandler(d => { onTimeArrivalMap = (d && d.map) || {}; if (ROWS.length) applyFilters(); })
        .withFailureHandler(e => console.warn('OTA load failed:', e));
      if (forceRefresh) runner.forceRefreshOnTimeArrivalData(); else runner.getOnTimeArrivalData(false);
    }

    function onError(err) {
      document.getElementById('stateMsg').style.display = 'none';
      const box = document.getElementById('errorBox');
      box.style.display = 'block';
      box.textContent = 'Could not load data: ' + (err && err.message ? err.message : err);
    }

    function onData(data) {
      HEADERS = data.headers;
      ROWS = data.rows;
      document.getElementById('subtitle').textContent =
        ROWS.length + ' trips · updated ' + new Date(data.generatedAt).toLocaleString();
      buildHeader();
      populateFilterOptions();
      applyFilters();
      document.getElementById('controls').style.display = 'flex';
      document.getElementById('statPills').style.display = 'flex';
    }

    // ---- Header / sorting --------------------------------------------
    function buildHeader() {
      const tr = document.getElementById('headerRow');
      tr.innerHTML = '';
      
      HEADERS.forEach((h, i) => {
        const th = document.createElement('th');
        th.innerHTML = escapeHtml(h || '(unnamed)') + '<span class="arrow" data-idx="' + i + '"></span>';
        th.addEventListener('click', () => toggleSort(i));
        tr.appendChild(th);

        // แทรกคอลัมน์ Late Cut ถัดจาก Status (COL.STATUS = 14)
        if (i === COL.STATUS) {
          const thLateCut = document.createElement('th');
          thLateCut.innerHTML = 'Late Cut<span class="arrow" data-idx="lateCut"></span>';
          thLateCut.addEventListener('click', () => toggleSort('lateCut'));
          tr.appendChild(thLateCut);
        }
      });

      // Extra OTA column
      const thOta = document.createElement('th');
      thOta.innerHTML = 'OTA Status<span class="arrow" data-idx="ota"></span>';
      thOta.addEventListener('click', () => toggleSort('ota'));
      tr.appendChild(thOta);
    }

    function toggleSort(idx) {
      if (sortCol === idx) { sortDir = -sortDir; } else { sortCol = idx; sortDir = 1; }
      document.querySelectorAll('.arrow').forEach(a => a.textContent = '');
      const arrow = document.querySelector('.arrow[data-idx="' + idx + '"]');
      if (arrow) arrow.textContent = sortDir === 1 ? '▲' : '▼';
      renderRows();
    }

    // ---- Filters -----------------------------------------------------
    function populateFilterOptions() {
      fillSelect('filterCategory', uniqueValues(COL.TRIP_CATEGORY));
      fillSelect('filterRegion', uniqueValues(COL.REGION));
      fillSelect('filterStatus', uniqueValues(COL.STATUS));
    }

    function uniqueValues(colIdx) {
      const set = new Set();
      ROWS.forEach(r => { const v = r.cells[colIdx]; if (v) set.add(v); });
      return Array.from(set).sort();
    }

    function fillSelect(id, values) {
      const sel = document.getElementById(id);
      const current = sel.value;
      sel.innerHTML = sel.options[0].outerHTML;
      values.forEach(v => {
        const opt = document.createElement('option');
        opt.value = v; opt.textContent = v;
        sel.appendChild(opt);
      });
      sel.value = current;
    }

    function applyFilters() {
      const q = document.getElementById('searchInput').value.trim().toLowerCase();
      const category = document.getElementById('filterCategory').value;
      const region = document.getElementById('filterRegion').value;
      const status = document.getElementById('filterStatus').value;
      const ota = document.getElementById('filterOTA').value;

      FILTERED = ROWS.filter(r => {
        if (category && r.cells[COL.TRIP_CATEGORY] !== category) return false;
        if (region && r.cells[COL.REGION] !== region) return false;
        if (status && r.cells[COL.STATUS] !== status) return false;
        if (ota) { if (getOTACategory(r) !== ota) return false; }
        if (q) {
          const hay = searchable.map(i => (r.cells[i] || '')).join(' ').toLowerCase();
          if (!hay.includes(q)) return false;
        }
        return true;
      });

      renderStats();
      renderRows();
    }

    function renderStats() {
      const onTime = FILTERED.filter(r => (r.cells[COL.STATUS] || '').trim().toLowerCase() === 'on time' || (r.cells[COL.STATUS] || '').trim().toLowerCase() === 'on-time').length;
      const late = FILTERED.filter(r => (r.cells[COL.STATUS] || '').trim().toLowerCase() === 'late').length;
      const rc = FILTERED.filter(r => getOTACategory(r) === 'RC').length;
      const otaOnTime = FILTERED.filter(r => getOTACategory(r) === 'On time').length;
      const lhLate = FILTERED.filter(r => getOTACategory(r) === 'LH Late').length;
      const obLate = FILTERED.filter(r => getOTACategory(r) === 'OB Late').length;
      const pills = document.getElementById('statPills');
      pills.innerHTML =
        '<div class="pill total">Showing <b>' + FILTERED.length + '</b> of ' + ROWS.length + '</div>' +
        '<div class="pill ontime">On time <b>' + onTime + '</b></div>' +
        '<div class="pill late">Late <b>' + late + '</b></div>' +
        (rc ? '<div class="pill rc">RC <b>' + rc + '</b></div>' : '') +
        (otaOnTime ? '<div class="pill ontime">OTA On time <b>' + otaOnTime + '</b></div>' : '') +
        (lhLate ? '<div class="pill late">LH Late <b>' + lhLate + '</b></div>' : '') +
        (obLate ? '<div class="pill oblate">OB Late <b>' + obLate + '</b></div>' : '');
    }

    // ---- Rendering ---------------------------------------------------
    function renderRows() {
      let rows = FILTERED.slice();

      if (sortCol !== null) {
        rows.sort((a, b) => {
          let av, bv;
          if (sortCol === 'ota') {
            av = getOTACategory(a) || '';
            bv = getOTACategory(b) || '';
          } else if (sortCol === 'lateCut') {
            av = getLateCutDetail(a) || '';
            bv = getLateCutDetail(b) || '';
          } else {
            av = a.cells[sortCol] || '';
            bv = b.cells[sortCol] || '';
          }
          const an = parseFloat(av), bn = parseFloat(bv);
          let cmp;
          if (!isNaN(an) && !isNaN(bn) && av !== '' && bv !== '') {
            cmp = an - bn;
          } else {
            cmp = av.localeCompare(bv, undefined, { numeric: true, sensitivity: 'base' });
          }
          return cmp * sortDir;
        });
      }

      const tbody = document.getElementById('bodyRows');
      const frag = document.createDocumentFragment();

      rows.forEach(r => {
        const tr = document.createElement('tr');
        r.cells.forEach((val, i) => {
          const td = document.createElement('td');
          if (i === COL.STATUS && val) {
            const s = (val || '').trim().toLowerCase();
            td.innerHTML = '<span class="badge ' + (s === 'on time' || s === 'on-time' ? 'on-time' : 'late') + '">' + escapeHtml(val) + '</span>';
          } else if (i === COL.TRIP_CATEGORY && val) {
            td.innerHTML = '<span class="chip ' + colorForCategory(val) + '">' + escapeHtml(val) + '</span>';
          } else if (i === COL.REGION && val) {
            td.innerHTML = '<span class="region-tag">' + escapeHtml(val) + '</span>';
          } else if (i === COL.ROUTE) {
            if (r.routeLink) {
              const a = document.createElement('a');
              a.href = r.routeLink; a.target = '_blank'; a.rel = 'noopener';
              a.className = 'route-link'; a.textContent = val || 'View route ↗';
              td.appendChild(a);
            } else if (val) {
              td.textContent = val;
            } else {
              td.innerHTML = '<span class="empty-cell">—</span>';
            }
          } else if (val === '' || val == null) {
            td.innerHTML = '<span class="empty-cell">—</span>';
          } else {
            td.textContent = val;
          }
          tr.appendChild(td);

          // แทรกคอลัมน์ Late Cut ในฝั่งข้อมูล
          if (i === COL.STATUS) {
            const tdLateCut = document.createElement('td');
            const lc = getLateCutDetail(r);
            if (lc) {
              tdLateCut.innerHTML = '<span class="badge late-cut">' + escapeHtml(lc) + '</span>';
            } else {
              tdLateCut.innerHTML = '<span class="empty-cell">—</span>';
            }
            tr.appendChild(tdLateCut);
          }
        });

        // OTA column
        const tdOta = document.createElement('td');
        tdOta.innerHTML = otaBadgeHtml(getOTACategory(r));
        tr.appendChild(tdOta);
        frag.appendChild(tr);
      });

      tbody.innerHTML = '';
      tbody.appendChild(frag);
      document.getElementById('stateMsg').style.display = 'none';
      document.getElementById('dataTable').style.display = rows.length ? 'table' : 'none';
      if (!rows.length) {
        document.getElementById('stateMsg').style.display = 'block';
        document.getElementById('stateMsg').innerHTML = '<div>No trips match your filters.</div>';
      }
    }

    function truckLoaderHtml(text) {
      return '<div class="truck-loader"><div class="truck-track"><div class="road-line"></div>' +
        '<svg class="truck-svg" viewBox="0 0 96 60" xmlns="http://www.w3.org/2000/svg">' +
        '<rect x="2" y="16" width="50" height="26" rx="4" fill="#EE4D2D"/>' +
        '<path d="M52 22 L72 22 L84 34 L84 42 L52 42 Z" fill="#FF7A45"/>' +
